get_eia_sep <- function(state, abbr){
    library(tidyverse)
    library(stringr)
    library(httr2)
    library(rvest)
    library(fs)
  
    state_formatted <- str_to_lower(state) |> str_replace_all("\\s", "")
    
    dir_name <- fs::path("data", "mp02")
    file_name <- fs::path(dir_name, state_formatted)
    
    fs::dir_create(dir_name, recurse=TRUE)
    
    if(!fs::file_exists(file_name)){
        BASE_URL <- "https://www.eia.gov"
        RESPONSE <- request(BASE_URL) |> 
          req_url_path("electricity", 
                       "state", 
                       state_formatted) |>
          req_headers(
            `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0"
            ) |>   
          req_retry(
                max_tries = 4, 
          ) |>
          req_perform(file_name)
    
        resp_check_status(RESPONSE)
    }
    
    TABLE <- read_html(file_name) |> 
        html_element("table") |> 
        html_table() |>
        mutate(Item = str_to_lower(Item))
    
    if("U.S. rank" %in% colnames(TABLE)){
        TABLE <- TABLE |> rename(Rank = `U.S. rank`)
    }
    
    CO2_MWh <- TABLE |> 
        filter(Item == "carbon dioxide (lbs/mwh)") |>
        pull(Value) |> 
        str_replace_all(",", "") |>
        as.numeric()
    
    PRIMARY <- TABLE |> 
        filter(Item == "primary energy source") |> 
        pull(Rank)
    
    RATE <- TABLE |>
        filter(Item == "average retail price (cents/kwh)") |>
        pull(Value) |>
        as.numeric()
    
    GENERATION_MWh <- TABLE |>
        filter(Item == "net generation (megawatthours)") |>
        pull(Value) |>
        str_replace_all(",", "") |>
        as.numeric()
    
    data.frame(CO2_MWh               = CO2_MWh, 
               primary_source        = PRIMARY,
               electricity_price_MWh = RATE * 10, # / 100 cents to dollars &
               # * 1000 kWh to MWH 
               generation_MWh        = GENERATION_MWh, 
               state                 = state, 
               abbreviation          = abbr
    )
}

EIA_SEP_REPORT <- map2(state.name, state.abb, get_eia_sep) |> list_rbind()

load_ntd_energy <- function(){
  library(readxl)
  library(httr2)
  library(tidyverse)
  library(fs)
  
  # Create 'data/mp02' directory if not already present
  DATA_DIR <- fs::path("data", "mp02")
  fs::dir_create(DATA_DIR, recurse=TRUE)

  NTD_ENERGY_FILE <- fs::path(DATA_DIR, "2025_ntd_energy.xlsx")

  if(!fs::file_exists(NTD_ENERGY_FILE)){
    library(httr2)
  
    request("https://www.transit.dot.gov/") |>
      req_url_path("sites",
                   "fta.dot.gov",
                   "files",
                   "2026-09",
                   "2025 Energy Consumption_260806.xlsx") |> 
      req_headers(
        `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0") |> 
      req_perform(NTD_ENERGY_FILE)
  }
  
  if(file.info(NTD_ENERGY_FILE)$size < 1e4){
        cat("I was unable to download the NTD Energy File. Please try again.\n")
        stop("Download failed. Please delete the corrupted file by running `unlink('data/mp02/2025_ntd_energy.xlsx')` and try again.")
    }

  to_numeric_fill_0 <- function(x){
    x <- if_else(x == "-", NA, x)
    replace_na(as.numeric(x), 0)
  }

  read_excel(NTD_ENERGY_FILE) |> 
    select(-c(`Reporter Type`, 
              `Reporting Module`, 
              `Gallons of Other Fuel`, 
              `Other Fuel Description`)) |>
    mutate(across(-c(`Agency Name`, 
                     `Mode`,
                     `TOS`), 
                  to_numeric_fill_0)) |>
    group_by(`NTD ID`, `Mode`, `Agency Name`) |>
    summarize(across(where(is.numeric), sum), 
              .groups = "keep") |>
    mutate(ENERGY = sum(c_across(c(where(is.numeric))))) |>
    filter(ENERGY > 0) |>
    select(-ENERGY) |>
    ungroup()
}

NTD_ENERGY <- load_ntd_energy()

load_ntd_service <- function(){
  library(httr2)
  library(tidyverse)
  library(fs)
  
  # Create 'data/mp02' directory if not already present
  DATA_DIR <- fs::path("data", "mp02")
  fs::dir_create(DATA_DIR, recurse=TRUE)

  NTD_SERVICE_FILE <- fs::path(DATA_DIR, "2025_ntd_service.csv")

  if(!fs::file_exists(NTD_SERVICE_FILE)){
    library(httr2)
  
    request("https://data.transportation.gov/resource/4fir-qbim.csv") |>
      req_url_query(`$query`=paste(
        "SELECT  `_5_digit_ntd_id` AS `NTD_ID`", 
        "`max_agency` AS `AGENCY`",
        "`max_reporter_type` AS `reporter_type`",
        "`max_organization_type` AS `organization_type`",
        "`max_city` AS `city`",
        "`max_state` AS `state`",
        "`mode`",
        "`type_of_service`",
        "`max_primary_uza_code` as `uza_code`",
        "`max_primary_uza_name` as `uza_name`",
        "`max_primary_uza_area_sq_miles` as `uza_sq_miles`",
        "`max_primary_uza_population` as `uza_population`",
        "`max_service_area_sq_miles` as `service_area_sq_miles`",
        "`max_service_area_population` as `service_area_population`",
        "`questionable_record` as `flagged_likely_inaccurate`",
        "`sum_unlinked_passenger_trips_upt` as `total_upt`",
        "`sum_passenger_miles` WHERE caseless_one_of(`report_year`, '2025') limit 50000", 
        sep=",")) |>
      req_headers(
        `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0") |> 
      req_perform(NTD_SERVICE_FILE)
  }
  
  if(file.info(NTD_SERVICE_FILE)$size < 1e4){
        cat("I was unable to download the NTD Service File. Please try again.\n")
        stop("Download failed. Please delete the corrupted file by running `unlink('data/mp02/2025_ntd_service.xlsx')` and try again.")
    }

  read_csv(NTD_SERVICE_FILE) |>
    mutate(flagged_likely_inaccurate = !is.na(flagged_likely_inaccurate))
}

NTD_SERVICE <- load_ntd_service()

# Calculate total fuel consumption for each transit agency
agency_fuel <- NTD_ENERGY |>
  group_by(`NTD ID`, `Agency Name`) |>
  summarize(
    across(starts_with("Gallons of"), ~sum(.x, na.rm = TRUE)),
    .groups = "drop"

Estimating CO2 Emissions

Code 
# Approximate combustion coefficients in pounds of CO2
# per reported gallon or gallon equivalent.
# Sources: EIA CO2 Emissions Coefficients and DOE AFDC

transit_emissions <- transit_joined |>
  mutate(
    # Electricity: convert kWh to MWh
    electricity_kwh =
      coalesce(`Kilowatt hours of Electric Battery`, 0) +
      coalesce(`Kilowatt hours of Electric Propulsion`, 0),

    electricity_co2_lb =
      electricity_kwh / 1000 * CO2_MWh,

    # Direct combustion emissions
    combustion_co2_lb =
      coalesce(`Gallons of Diesel Fuel`, 0) * 22.45 +
      coalesce(`Gallons of Gasoline`, 0) * 18.73 +
      coalesce(`Gallons of Bio-Diesel`, 0) * 22.45 +
      coalesce(`Gallons of Bunker Fuel`, 0) * 24.78 +
      coalesce(`Gallons of Kerosene`, 0) * 21.78 +
      coalesce(`Gallons of Compressed Natural Gas`, 0) * 13.33 +
      coalesce(`Gallons of Liquified Nat Gas`, 0) * 8.80 +
      coalesce(`Gallons of Liquified Petroleum Gas`, 0) * 12.68 +
      coalesce(`Gallons of Ethanol`, 0) * 0 +
      coalesce(`Gallons of Hydrogen`, 0) * 0,

    # Do not treat unmatched records as zero emissions
    total_co2_lb = if_else(
      !is.na(`Agency Name`) & !is.na(CO2_MWh),
      electricity_co2_lb + combustion_co2_lb,
      NA_real_
    )
  )

  group_by(NTD_ID) |>

    AGENCY = first(AGENCY),
    modes = n(),

    total_upt = sum(total_upt, na.rm = TRUE),

    total_passenger_miles =
      if (all(is.na(passenger_miles))) {
        NA_real_
      } else {
        sum(passenger_miles, na.rm = TRUE)
      },

    total_co2_lb =
      if (all(is.na(total_co2_lb))) {
        NA_real_
      } else {
        sum(total_co2_lb, na.rm = TRUE)
      },

    missing_emissions_modes = sum(is.na(total_co2_lb)),
    missing_mile_rows = sum(missing_mile_rows),

    # Pounds CO2 per unlinked passenger trip

      total_upt > 0 & missing_emissions_modes == 0,

    co2_per_passenger_mile = if_else(
      !is.na(total_passenger_miles) &
        total_passenger_miles > 0 &
        missing_emissions_modes == 0 &
        missing_mile_rows == 0,

      NA_real_
    )
  )

GTA IV Green Transit Awards: Official Press Release

FOR IMMEDIATE RELEASE - November 19, 2026

Seattle Center Monorail Takes the Top Spot in the GTA IV Green Transit Awards

Seattle Center Monorail leads in low estimated CO2 emissions, while SeaStreak, LLC receives the Vincidoo Award for the highest emissions per passenger trip.

November 19, 2026 — Which public transit agency is the greenest in America? The GTA IV Green Transit Awards set out to answer this question by comparing the estimated CO2 emissions of 283 eligible transit agencies across the United States.

Using publicly available transportation and energy data, the awards recognize agencies with the lowest and highest estimated emissions per passenger trip. The goal is to highlight how environmental performance can vary across public transportation systems.

Greenest Transit Agency Award: Seattle Center Monorail

Seattle Center Monorail takes home the Greenest Transit Agency Award with just 0.058 pounds of CO2 per passenger trip.

To determine the winner, we compared each agency’s estimated CO2 emissions against its annual passenger trips. Only agencies with at least one million annual trips and sufficient emissions data qualified. For comparison, the median eligible agency produced approximately 4.301 pounds of CO2 per trip. Seattle Center Monorail’s emissions were 98.7% lower than the median, putting it in first place. With over 2.29 million annual passenger trips, Seattle Center Monorail shows how electric public transportation can serve large numbers of passengers while keeping estimated emissions relatively low. For Seattle Center Monorail, staying on track also means keeping its estimated carbon footprint surprisingly small.

Vincidoo Award: SeaStreak, LLC

Not every agency made the green list. The Vincidoo Award, our newly introduced “Worst Of” category, goes to SeaStreak, LLC, which recorded the highest estimated CO2 emissions per passenger trip among eligible agencies. SeaStreak recorded approximately 43.77 pounds of CO2 per passenger trip, more than 10 times the median of 4.30 pounds among eligible agencies.

SeaStreak certainly made waves in this year’s rankings but this award comes with an important plot twist. But there is more to the story. SeaStreak operates ferry services, which may cover much longer distances than a typical bus or subway trip. Since our rankings are based on emissions per trip rather than distance traveled, this does not necessarily mean SeaStreak is the least environmentally friendly transit agency overall. Instead, it highlights why different types of transportation should be considered carefully when comparing environmental impact.

Bonus Award: The Plot Twist Award — Going the Extra Mile

And this year’s biggest plot twist goes to Academy Lines, Inc. (Academy)!

Academy ranked 271st in estimated CO2 emissions per passenger trip but jumped to 23rd when emissions were measured per passenger mile, an improvement of 248 positions.

To determine the winner, we ranked agencies with valid results on both measures and identified the largest improvement between the two rankings. Academy recorded approximately 16.719 pounds of CO2 per trip, compared with 0.396 pounds per passenger mile.

For reference, the median rank change among the 282 comparable agencies was -3.5 positions.

The lesson? Sometimes the biggest environmental plot twist isn’t about changing the transportation system. It’s about changing how we measure it. A longer trip may generate more emissions overall while remaining relatively efficient per passenger mile.

Behind the Awards (How We Selected the Winners)

To make the comparisons, we combined publicly available data on transit ridership, fuel consumption, electricity use, and state electricity emissions. We estimated each agency’s CO2 emissions based on its reported fuel and electricity consumption, then compared those emissions with the number of passenger trips and miles traveled. The Greenest Transit Agency and Vincidoo Awards were based on emissions per passenger trip, while the Plot Twist Award compared agency rankings using both measures.

It is important to recognize that these results are estimates, not exact measurements. Some agencies had incomplete data, and certain fuel calculations required assumptions. The analysis also does not include every environmental impact associated with producing and operating transit vehicles.

Ultimately, the GTA IV Green Transit Awards offer a way to better understand the environmental differences across public transportation systems. While the winners represent opposite ends of the rankings, both remind us that evaluating greener transportation requires looking beyond the total amount of energy used.

Data Relationship Diagram

The following diagram shows the three datasets used in this analysis, their columns and data types, and the relationships used to combine them.

How the Datasets Were Connected

NTD_SERVICE and NTD_ENERGY: These datasets were connected using the agency ID and transit mode. Because the agency IDs had different data types, the energy dataset numeric ID was converted into a five-character ID. Service records were also grouped by agency and mode before joining to avoid duplicating fuel consumption.

NTD_SERVICE and EIA_SEP_REPORT: These datasets were connected using the state abbreviations. Multiple transit agencies can belong to the same state, while each state has one electricity emissions factor in the EIA dataset.

These relationships allowed us to combine ridership, energy consumption, and state electricity emissions information to estimate transit agency CO2 emissions.

Sources: Federal Transit Administration National Transit Database and U.S. Energy Information Administration State Electricity Profiles.
